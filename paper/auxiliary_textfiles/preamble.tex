% ===== CUSTOM COMMANDS =====

% Chapter command without numbering but in TOC
\newcommand{\chap}[1]{%
  \chapter*{#1}%
  \addcontentsline{toc}{chapter}{#1}%
}

% Section command without numbering but in TOC
\newcommand{\sect}[1]{%
  \section*{#1}%
  \addcontentsline{toc}{section}{#1}%
}

% Subsection command without numbering
\newcommand{\subsect}[1]{%
  \subsection*{#1}%
}

% Roman numerals for papers (if needed)
\newcommand{\I}{\textsc{i}}
\newcommand{\II}{\textsc{ii}}
\newcommand{\III}{\textsc{iii}}
\newcommand{\IV}{\textsc{iv}}

  showstringspaces=false,
  showtabs=false,
  frame=single,
  tabsize=2,

}
